// WBSQUETE · placar dos 6 confrontos na tela do celular (app Scriptable)
// Tela bloqueada (retangular): os 6 placares em 3 linhas, o seu em destaque.
// Tela de início (médio ou grande): quadrinhos com os logos, igual ao widget da NBA.
// Parâmetro do widget: o id do seu time (1 a 12). Sem parâmetro = Iguana (10).
// 1 Tubas · 2 Baianas · 3 Panam Bay · 4 Seridó · 5 Icers · 6 Necks · 7 Judazz
// 8 Verdes · 9 Osasco · 10 Iguana · 11 Luiz ☀️ · 12 Pompeia

const LIGA = "1009936782", TEMP = 2027
const URL_LIGA = `https://lm-api-reads.fantasy.espn.com/apis/v3/games/fba/seasons/${TEMP}/segments/0/leagues/${LIGA}`
const PAINEL = "https://lucascm4.github.io/wbsquete/"
const SIG = {1:"TUB",2:"BAI",3:"PAN",4:"SER",5:"ICE",6:"NEC",7:"JUD",8:"VER",9:"OSA",10:"IGU",11:"LUZ",12:"POM"}
const LOGO = {1:"tubas",2:"baianas",3:"panambay",4:"serido",5:"icers",6:"necks",7:"judazz",8:"verdes",9:"osasco",10:"iguana",11:"diabo",12:"pompeia"}

const MEU = parseInt((args.widgetParameter || "10").trim()) || 10

// ---------- o dia de jogo da ESPN segue o horário de Nova York ----------
function spHoje() {
  const ny = new Date().toLocaleDateString("en-CA", { timeZone: "America/New_York" })
  const [a, m, d] = ny.split("-").map(Number)
  const dias = Math.round((Date.UTC(a, m - 1, d) - Date.UTC(2026, 9, 20)) / 864e5) // 20/10/2026 = dia 1
  return Math.max(1, dias + 1)
}
function semanaDe(sp) { return sp < 7 ? { k: 1, s0: 1, s1: 6 } : { k: 2 + Math.floor((sp - 7) / 7), s0: 7 + Math.floor((sp - 7) / 7) * 7, s1: 13 + Math.floor((sp - 7) / 7) * 7 } }

async function json(url) { const r = new Request(url); r.timeoutInterval = 20; return await r.loadJSON() }

// soma o que os titulares (vaga < 12) fizeram num dia: [PTS,REB,AST,STL,BLK,3PM,TO,FGM,FGA,FTM,FTA]
function somaDia(time, sp, tot) {
  for (const e of time.roster.entries) {
    if (e.lineupSlotId >= 12) continue
    const st = (e.playerPoolEntry.player.stats || []).find(s => s.scoringPeriodId === sp && s.statSourceId === 0 && s.statSplitTypeId === 5)
    if (!st || !st.stats || !st.stats["42"]) continue
    const s = st.stats
    ;[0, 6, 3, 2, 1, 17, 11, 13, 14, 15, 16].forEach((id, i) => tot[i] += (s[id] || 0))
  }
}
// placar em categorias (FG% FT% 3PM REB AST STL BLK TO PTS; TO menos é melhor)
function placar(A, B) {
  const v = x => [x[8] ? x[7] / x[8] : 0, x[10] ? x[9] / x[10] : 0, x[5], x[1], x[2], x[3], x[4], -x[6], x[0]]
  const a = v(A), b = v(B); let w = 0, l = 0, t = 0
  a.forEach((x, i) => { const y = b[i]; if (Math.abs(x - y) < 1e-9) t++; else if (x > y) w++; else l++ })
  return { w, l, t }
}

async function dados() {
  const cur = spHoje(), sem = semanaDe(cur)
  const sb = await json(`${URL_LIGA}?view=mMatchupScore&scoringPeriodId=${cur}`)
  const jogos = sb.schedule.filter(m => m.matchupPeriodId === sem.k && m.home && m.away).map(m => ({ a: m.away.teamId, b: m.home.teamId }))
  const tot = {}; for (let t = 1; t <= 12; t++) tot[t] = Array(11).fill(0)
  const fim = Math.min(cur, sem.s1), dias = []
  for (let sp = sem.s0; sp <= fim; sp++) dias.push(json(`${URL_LIGA}?view=mRoster&scoringPeriodId=${sp}`).then(r => [sp, r]))
  for (const [sp, r] of await Promise.all(dias)) r.teams.forEach(t => { if (tot[t.id]) somaDia(t, sp, tot[t.id]) })
  jogos.forEach(j => { if (j.b === MEU) { const x = j.a; j.a = j.b; j.b = x } j.p = placar(tot[j.a], tot[j.b]) })
  jogos.sort((x, y) => (y.a === MEU) - (x.a === MEU)) // o seu primeiro
  return { jogos, dia: fim - sem.s0 + 1, ndias: sem.s1 - sem.s0 + 1, k: sem.k }
}

// logos: baixa do painel uma vez e guarda no celular
const FM = FileManager.local(), PASTA = FM.joinPath(FM.documentsDirectory(), "wbsquete_logos")
async function logo(t) {
  try {
    if (!FM.fileExists(PASTA)) FM.createDirectory(PASTA)
    const f = FM.joinPath(PASTA, LOGO[t] + ".png")
    if (FM.fileExists(f)) return FM.readImage(f)
    const r = new Request(`${PAINEL}img/l/${LOGO[t]}.webp`); r.timeoutInterval = 10
    const img = await r.loadImage(); FM.writeImage(f, img); return img
  } catch (e) { return null }
}

const AMARELO = new Color("#ffd400"), VERM = new Color("#ff5a5a"), VERDE = new Color("#3ddc84"), CINZA = new Color("#9a8f8f")
const txt = p => `${p.w}-${p.l}${p.t ? "-" + p.t : ""}`

async function quadro(pai, j, tam) {
  const c = pai.addStack(); c.layoutHorizontally(); c.centerAlignContent()
  c.backgroundColor = j.a === MEU ? new Color("#3a2a08") : new Color("#1c1c1e"); c.cornerRadius = 12
  c.setPadding(tam.pad, 6, tam.pad, 6); c.size = new Size(tam.w, tam.h)
  const lado = async t => {
    const s = c.addStack(); s.layoutVertically(); s.centerAlignContent()
    const img = await logo(t)
    if (img) { const im = s.addImage(img); im.imageSize = new Size(tam.logo, tam.logo); im.cornerRadius = tam.logo / 2 }
    const n = s.addText(SIG[t]); n.font = Font.semiboldSystemFont(tam.sig); n.textColor = Color.white()
  }
  await lado(j.a)
  c.addSpacer()
  const p = c.addText(txt(j.p)); p.font = Font.heavySystemFont(tam.pl); p.minimumScaleFactor = 0.7
  p.textColor = j.p.w > j.p.l ? VERDE : j.p.w < j.p.l ? VERM : AMARELO
  c.addSpacer()
  await lado(j.b)
}

async function montar() {
  const fam = config.widgetFamily || "large"
  const w = new ListWidget(); w.url = PAINEL
  w.refreshAfterDate = new Date(Date.now() + 10 * 60 * 1000)
  let d
  try { d = await dados() } catch (e) {
    const t = w.addText(fam.startsWith("accessory") ? "WBSQUETE: sem sinal" : "WBSQUETE\nNão consegui ler a ESPN agora."); t.font = Font.semiboldSystemFont(12)
    return w
  }
  const J = d.jogos

  if (fam === "accessoryInline") { w.addText(`${SIG[J[0].a]} ${txt(J[0].p)} ${SIG[J[0].b]}`); return w }
  if (fam === "accessoryCircular") {
    w.addAccessoryWidgetBackground = true
    const a = w.addText(SIG[J[0].a]); a.font = Font.boldSystemFont(10); a.centerAlignText()
    const b = w.addText(txt(J[0].p)); b.font = Font.heavySystemFont(15); b.centerAlignText(); b.minimumScaleFactor = 0.6
    return w
  }
  if (fam === "accessoryRectangular") {
    for (let i = 0; i < 6; i += 2) {
      const l = w.addStack(); l.spacing = 8
      ;[J[i], J[i + 1]].forEach(j => {
        if (!j) return
        const t = l.addText(`${SIG[j.a]} ${txt(j.p)} ${SIG[j.b]}`)
        t.font = j.a === MEU ? Font.heavySystemFont(12) : Font.mediumSystemFont(11); t.lineLimit = 1; t.minimumScaleFactor = 0.6
      })
    }
    return w
  }

  // ----- tela de início: quadrinhos com logo, igual ao da NBA -----
  w.backgroundColor = new Color("#0b0b0c"); w.setPadding(12, 12, 12, 12)
  const topo = w.addStack(); topo.centerAlignContent()
  const tt = topo.addText("WBSQUETE"); tt.font = Font.heavySystemFont(13); tt.textColor = AMARELO
  topo.addSpacer()
  const sub = topo.addText(`SEMANA ${d.k} · DIA ${d.dia}/${d.ndias}`); sub.font = Font.semiboldSystemFont(10); sub.textColor = CINZA
  w.addSpacer(8)
  const grande = fam === "large" || fam === "extraLarge"
  const tam = grande ? { w: 146, h: 92, pad: 8, logo: 34, sig: 11, pl: 24 } : { w: 146, h: 36, pad: 3, logo: 18, sig: 8, pl: 15 }
  const lista = fam === "small" ? J.slice(0, 1) : J
  for (let i = 0; i < lista.length; i += 2) {
    const linha = w.addStack(); linha.spacing = 8
    await quadro(linha, lista[i], tam)
    if (lista[i + 1] && fam !== "small") await quadro(linha, lista[i + 1], tam)
    w.addSpacer(grande ? 8 : 4)
  }
  w.addSpacer()
  const at = w.addText("atualizado " + new Date().toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" }))
  at.font = Font.systemFont(8); at.textColor = CINZA
  return w
}

const widget = await montar()
if (config.runsInWidget) Script.setWidget(widget)
else await widget.presentLarge()
Script.complete()
